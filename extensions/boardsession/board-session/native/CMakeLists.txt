cmake_minimum_required(VERSION 3.21)
project(board_session LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This release builds Linux native artifacts")
endif()
set(WOLFSSH_ROOT "" CACHE PATH "wolfSSH install prefix (X.509 and terminal enabled)")
set(WOLFSSL_ROOT "" CACHE PATH "wolfSSL install prefix used by wolfSSH")
set(NODE_INCLUDE_DIR "" CACHE PATH "Node.js headers containing node_api.h")
set(ARTIFACT_DIR "" CACHE PATH "Platform artifact directory")
foreach(ENTRY WOLFSSH_INCLUDE WOLFSSL_INCLUDE WOLFSSH_LIBRARY WOLFSSL_LIBRARY)
  unset(${ENTRY} CACHE)
endforeach()
find_path(WOLFSSH_INCLUDE wolfssh/ssh.h PATHS "${WOLFSSH_ROOT}/include" NO_DEFAULT_PATH NO_CACHE REQUIRED)
find_path(WOLFSSL_INCLUDE wolfssl/options.h PATHS "${WOLFSSL_ROOT}/include" NO_DEFAULT_PATH NO_CACHE REQUIRED)
find_library(WOLFSSH_LIBRARY wolfssh PATHS "${WOLFSSH_ROOT}/lib" NO_DEFAULT_PATH NO_CACHE REQUIRED)
find_library(WOLFSSL_LIBRARY wolfssl PATHS "${WOLFSSL_ROOT}/lib" NO_DEFAULT_PATH NO_CACHE REQUIRED)
add_library(board_session MODULE addon.cc)
target_compile_features(board_session PRIVATE cxx_std_17)
target_compile_definitions(board_session PRIVATE NAPI_VERSION=8 WOLFSSH_CERTS WOLFSSH_TERM)
target_compile_options(board_session PRIVATE -Wall -Wextra -Werror)
target_include_directories(board_session PRIVATE "${NODE_INCLUDE_DIR}" "${WOLFSSH_INCLUDE}" "${WOLFSSL_INCLUDE}")
target_link_options(board_session PRIVATE "-Wl,--disable-new-dtags")
target_link_libraries(board_session PRIVATE "${WOLFSSH_LIBRARY}" "${WOLFSSL_LIBRARY}")
set_target_properties(board_session PROPERTIES PREFIX "" SUFFIX ".node"
  LIBRARY_OUTPUT_DIRECTORY "${ARTIFACT_DIR}" BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
# Copy actual files under every SONAME as npm does not preserve symlinks.
file(GLOB RUNTIME_LIBRARIES "${WOLFSSH_ROOT}/lib/libwolfssh.so*" "${WOLFSSL_ROOT}/lib/libwolfssl.so*")
foreach(LIBRARY IN LISTS RUNTIME_LIBRARIES)
  get_filename_component(NAME "${LIBRARY}" NAME)
  configure_file("${LIBRARY}" "${ARTIFACT_DIR}/${NAME}" COPYONLY)
endforeach()
